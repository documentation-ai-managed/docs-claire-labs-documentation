<CodeGroup tabs="JavaScript,Python">
  ```javascript
// Quick start example
const claireLabs = require('claire-labs-sdk');

const client = new claireLabs.Client({
  apiKey: 'your-api-key-here'
});

// Assess a student submission
const result = await client.assess({
  content: 'Student essay text...',
  rubric: 'essay-rubric-1'
});

console.log(result.score, result.feedback);
  ```
  ```python
from claire_labs import Client

client = Client(api_key='your-api-key-here')

# Assess submission
result = client.assess(
    content='Student essay text...',
    rubric='essay-rubric-1'
)

print(result.score, result.feedback)
  ```
